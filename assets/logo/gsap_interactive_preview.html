// FreelanceOS — Signature GSAP SVG Animation
const tl = gsap.timeline({ defaults: { ease: "power3.out" } });

// 1. Measure and prime SVG stroke paths
const stem = document.querySelector("#anim-stem");
const crossbar = document.querySelector("#anim-crossbar");
const stemLen = stem.getTotalLength();
const crossLen = crossbar.getTotalLength();

// 2. Hero ignition sequence
tl.fromTo([stem, crossbar], 
  { strokeDasharray: (i) => i === 0 ? stemLen : crossLen, 
    strokeDashoffset: (i) => i === 0 ? stemLen : crossLen },
  { strokeDashoffset: 0, duration: 1.2, stagger: 0.18 }
)
.from("#anim-spark", {
  scale: 0,
  rotation: -90,
  duration: 0.9,
  ease: "back.out(2)",
  transformOrigin: "348px 256px"
}, "-=0.6")
.from("#anim-aura", {
  opacity: 0,
  scale: 0.6,
  duration: 1.0,
  transformOrigin: "center"
}, "-=0.8")
.from("#anim-dot", {
  scale: 0,
  opacity: 0,
  duration: 0.4,
  ease: "back.out(3)"
}, "-=0.5");
